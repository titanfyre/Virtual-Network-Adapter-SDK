cmake_minimum_required(VERSION 3.20)
cmake_policy(SET CMP0091 NEW)
project(TitanVirtualNetwork VERSION 0.1.1 LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
 message(FATAL_ERROR "Build with Visual Studio C++ tools on Windows x64.")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
option(TRACE_ASAN "Instrument all user-mode targets with AddressSanitizer" OFF)
add_compile_definitions(UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
add_compile_options(/W4 /WX /utf-8 /permissive- /EHsc /guard:cf)
add_link_options(/guard:cf /DYNAMICBASE /NXCOMPAT /HIGHENTROPYVA)
if(TRACE_ASAN)
 add_compile_options(/fsanitize=address /Zi)
 add_link_options(/INCREMENTAL:NO /DEBUG)
endif()
add_library(NetworkAdapter SDK/NetworkAdapter.cpp)
target_link_libraries(NetworkAdapter PUBLIC setupapi winhttp crypt32)
add_executable(TitanVirtualNetwork WIN32 App/Main.cpp)
target_link_libraries(TitanVirtualNetwork PRIVATE NetworkAdapter user32 gdi32 comctl32 comdlg32 shell32)
target_link_options(TitanVirtualNetwork PRIVATE "/MANIFESTUAC:level='requireAdministrator' uiAccess='false'")
add_executable(NetworkAdapterCLI App/Console.cpp)
target_link_libraries(NetworkAdapterCLI PRIVATE NetworkAdapter)
add_executable(NetworkAdapterExample Examples/frames.cpp)
target_link_libraries(NetworkAdapterExample PRIVATE NetworkAdapter)
add_executable(NetworkAdapterTests Tests/ProtocolTests.cpp)
target_link_libraries(NetworkAdapterTests PRIVATE NetworkAdapter)
enable_testing()
add_test(NAME FrameParsingAndBounds COMMAND NetworkAdapterTests)

set(DRIVER_PACKAGE_DIRECTORY "" CACHE PATH "Opaque signed TitanVirtualNetwork INF/CAT/SYS package")
set(PACKAGE_RESOURCE "")
if(DRIVER_PACKAGE_DIRECTORY)
 enable_language(RC)
 foreach(name TitanVirtualNetwork.inf TitanVirtualNetwork.cat TitanVirtualNetwork.sys)
  if(NOT EXISTS "${DRIVER_PACKAGE_DIRECTORY}/${name}")
   message(FATAL_ERROR "Incomplete opaque driver package: ${name}")
  endif()
 endforeach()
 set(PACKAGE_RESOURCE "${CMAKE_CURRENT_BINARY_DIR}/DriverPackage.rc")
 file(WRITE "${PACKAGE_RESOURCE}" "101 RCDATA \"${DRIVER_PACKAGE_DIRECTORY}/TitanVirtualNetwork.inf\"\n102 RCDATA \"${DRIVER_PACKAGE_DIRECTORY}/TitanVirtualNetwork.cat\"\n103 RCDATA \"${DRIVER_PACKAGE_DIRECTORY}/TitanVirtualNetwork.sys\"\n")
endif()
add_library(TraceSetupCore Installer/Setup.cpp)
target_link_libraries(TraceSetupCore PUBLIC setupapi newdev wintrust crypt32 advapi32 ole32)
add_executable(NetworkAdapterSetup WIN32 Installer/Main.cpp ${PACKAGE_RESOURCE})
add_executable(NetworkAdapterSetupConsole Installer/Console.cpp ${PACKAGE_RESOURCE})
foreach(target NetworkAdapterSetup NetworkAdapterSetupConsole)
 target_link_libraries(${target} PRIVATE TraceSetupCore user32 gdi32)
 target_link_options(${target} PRIVATE "/MANIFESTUAC:level='requireAdministrator' uiAccess='false'")
endforeach()

foreach(target TitanVirtualNetwork NetworkAdapterCLI NetworkAdapterExample NetworkAdapterTests NetworkAdapterSetup NetworkAdapterSetupConsole)
 set_target_properties(${target} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
endforeach()

add_executable(NetworkInstallerTests Tests/Installer.cpp)
target_link_libraries(NetworkInstallerTests PRIVATE TraceSetupCore)
set_target_properties(NetworkInstallerTests PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
add_test(NAME InstallerIdentityAndUnsignedRefusal COMMAND NetworkInstallerTests)
